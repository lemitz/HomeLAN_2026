```go
//go:build windows

package utils

import (
	"syscall"
	"unsafe"
)

func HasEnoughSpace(path string, required int64) bool {
	kernel32 := syscall.NewLazyDLL("kernel32.dll")
	getDiskFreeSpaceEx := kernel32.NewProc("GetDiskFreeSpaceExW")

	var freeBytesAvailable uint64
	var totalNumberOfBytes uint64
	var totalNumberOfFreeBytes uint64

	pathPtr, _ := syscall.UTF16PtrFromString(path)

	ret, _, _ := getDiskFreeSpaceEx.Call(
		uintptr(unsafe.Pointer(pathPtr)),
		uintptr(unsafe.Pointer(&freeBytesAvailable)),
		uintptr(unsafe.Pointer(&totalNumberOfBytes)),
		uintptr(unsafe.Pointer(&totalNumberOfFreeBytes)),
	)

	if ret == 0 {
		return true
	}

	return int64(freeBytesAvailable) > required+100*1024*1024
}
```

```go
package utils

import (
	"syscall"
)

func HasEnoughSpace(path string, required int64) bool {
	var stat syscall.Statfs_t
	if err := syscall.Statfs(path, &stat); err != nil {
		return true
	}
	available := int64(stat.Bavail) * int64(stat.Bsize)
	return available > required+100*1024*1024
}
```

```go
package ws

import (
	"encoding/json"
	"log"
	"sync"
	"time"

	"github.com/gorilla/websocket"
	"homelan-server/internal/db"
)

const (
	writeWait      = 10 * time.Second
	pingPeriod     = 30 * time.Second
	maxMessageSize = 1024 * 1024
)

type Client struct {
	hub      *Hub
	conn     *websocket.Conn
	send     chan []byte
	userID   int64
	deviceID string
	mu       sync.Mutex
}

type Hub struct {
	db         *db.Database
	clients    map[string]*Client
	register   chan *Client
	unregister chan *Client
	broadcast  chan *BroadcastMessage
	mu         sync.RWMutex
}

type BroadcastMessage struct {
	RecipientID   int64
	Data          []byte
	ExcludeDevice string
}

type WSMessage struct {
	Type string          `json:"type"`
	Data json.RawMessage `json:"data"`
}

func NewHub(database *db.Database) *Hub {
	return &Hub{
		db:         database,
		clients:    make(map[string]*Client),
		register:   make(chan *Client),
		unregister: make(chan *Client),
		broadcast:  make(chan *BroadcastMessage, 256),
	}
}

func (h *Hub) Run() {
	for {
		select {
		case client := <-h.register:
			h.mu.Lock()
			h.clients[client.deviceID] = client
			h.mu.Unlock()
			log.Printf("📱 Client connected: device=%s user=%d", client.deviceID, client.userID)

		case client := <-h.unregister:
			h.mu.Lock()
			if _, ok := h.clients[client.deviceID]; ok {
				delete(h.clients, client.deviceID)
				close(client.send)
			}
			h.mu.Unlock()
			log.Printf("📴 Client disconnected: device=%s", client.deviceID)

		case msg := <-h.broadcast:
			h.handleBroadcast(msg)
		}
	}
}

func (h *Hub) handleBroadcast(msg *BroadcastMessage) {
	h.mu.RLock()
	defer h.mu.RUnlock()

	if msg.RecipientID == 0 {
		for deviceID, client := range h.clients {
			if deviceID != msg.ExcludeDevice {
				select {
				case client.send <- msg.Data:
				default:
				}
			}
		}
	} else {
		for deviceID, client := range h.clients {
			if client.userID == msg.RecipientID && deviceID != msg.ExcludeDevice {
				select {
				case client.send <- msg.Data:
				default:
				}
			}
		}
	}
}

func (h *Hub) SendToUser(userID int64, data []byte, excludeDevice string) {
	h.broadcast <- &BroadcastMessage{
		RecipientID:   userID,
		Data:          data,
		ExcludeDevice: excludeDevice,
	}
}

func (h *Hub) SendToDevice(deviceID string, data []byte) {
	h.mu.RLock()
	client, ok := h.clients[deviceID]
	h.mu.RUnlock()

	if ok {
		select {
		case client.send <- data:
		default:
		}
	}
}

func (h *Hub) RegisterClient(conn *websocket.Conn, userID int64, deviceID string) *Client {
	client := &Client{
		hub:      h,
		conn:     conn,
		send:     make(chan []byte, 256),
		userID:   userID,
		deviceID: deviceID,
	}

	h.register <- client
	return client
}

func (h *Hub) UnregisterClient(client *Client) {
	h.unregister <- client
}

func (h *Hub) GetOnlineDevices(userID int64) []string {
	h.mu.RLock()
	defer h.mu.RUnlock()

	var devices []string
	for deviceID, client := range h.clients {
		if client.userID == userID {
			devices = append(devices, deviceID)
		}
	}
	return devices
}

func (h *Hub) IsDeviceOnline(deviceID string) bool {
	h.mu.RLock()
	defer h.mu.RUnlock()
	_, ok := h.clients[deviceID]
	return ok
}
```

```go
package ws

import (
	"encoding/json"
	"log"
	"time"

	"github.com/gorilla/websocket"
)

func (c *Client) ReadPump() {
	defer func() {
		c.hub.UnregisterClient(c)
		c.conn.Close()
	}()

	c.conn.SetReadLimit(maxMessageSize)
	c.conn.SetReadDeadline(time.Now().Add(60 * time.Second))

	c.conn.SetPongHandler(func(string) error {
		c.conn.SetReadDeadline(time.Now().Add(60 * time.Second))
		return nil
	})

	for {
		_, message, err := c.conn.ReadMessage()
		if err != nil {
			if websocket.IsUnexpectedCloseError(err, websocket.CloseGoingAway, websocket.CloseAbnormalClosure) {
				log.Printf("WebSocket error: %v", err)
			}
			break
		}

		c.hub.db.UpdateDeviceLastSeen(c.deviceID)
		c.handleMessage(message)
	}
}

func (c *Client) WritePump() {
	ticker := time.NewTicker(pingPeriod)
	defer func() {
		ticker.Stop()
		c.conn.Close()
	}()

	for {
		select {
		case message, ok := <-c.send:
			c.conn.SetWriteDeadline(time.Now().Add(writeWait))
			if !ok {
				c.conn.WriteMessage(websocket.CloseMessage, []byte{})
				return
			}

			c.mu.Lock()
			err := c.conn.WriteMessage(websocket.TextMessage, message)
			c.mu.Unlock()

			if err != nil {
				return
			}

		case <-ticker.C:
			c.conn.SetWriteDeadline(time.Now().Add(writeWait))
			if err := c.conn.WriteMessage(websocket.PingMessage, nil); err != nil {
				return
			}
		}
	}
}

func (c *Client) handleMessage(data []byte) {
	var msg WSMessage
	if err := json.Unmarshal(data, &msg); err != nil {
		log.Printf("Invalid message format: %v", err)
		c.sendError("invalid_format", "Invalid message format")
		return
	}

	switch msg.Type {
	case "send_message":
		c.handleSendMessage(msg.Data)
	case "sync":
		c.handleSync(msg.Data)
	case "ack":
		c.handleAck(msg.Data)
	case "read":
		c.handleRead(msg.Data)
	case "delete":
		c.handleDelete(msg.Data)
	case "ping":
		c.handlePing()
	default:
		c.sendError("unknown_type", "Unknown message type: "+msg.Type)
	}
}

func (c *Client) handleSendMessage(data json.RawMessage) {
	var req struct {
		RecipientID      int64  `json:"recipient_id"`
		EncryptedContent []byte `json:"encrypted_content"`
		ContentType      string `json:"content_type"`
		FileID           string `json:"file_id,omitempty"`
		ExpiresAt        int64  `json:"expires_at,omitempty"`
	}

	if err := json.Unmarshal(data, &req); err != nil {
		c.sendError("invalid_data", "Invalid message data")
		return
	}

	message, err := c.hub.db.CreateMessage(
		c.userID,
		req.RecipientID,
		req.EncryptedContent,
		req.ContentType,
		req.FileID,
		req.ExpiresAt,
	)
	if err != nil {
		log.Printf("Error creating message: %v", err)
		c.sendError("db_error", "Failed to save message")
		return
	}

	c.sendJSON(map[string]interface{}{
		"type": "message_sent",
		"data": map[string]interface{}{
			"seq_id":     message.SeqID,
			"created_at": message.CreatedAt,
		},
	})

	msgData, _ := json.Marshal(map[string]interface{}{
		"type": "new_message",
		"data": map[string]interface{}{
			"seq_id":            message.SeqID,
			"sender_id":         message.SenderID,
			"encrypted_content": message.EncryptedContent,
			"content_type":      message.ContentType,
			"file_id":           message.FileID,
			"created_at":        message.CreatedAt,
			"expires_at":        message.ExpiresAt,
		},
	})

	c.hub.SendToUser(req.RecipientID, msgData, "")
	c.hub.SendToUser(c.userID, msgData, c.deviceID)

	log.Printf("📨 Message sent: seq=%d from=%d to=%d", message.SeqID, c.userID, req.RecipientID)
}

func (c *Client) handleSync(data json.RawMessage) {
	var req struct {
		SinceSeqID int64 `json:"since_seq_id"`
		Limit      int   `json:"limit,omitempty"`
	}

	if err := json.Unmarshal(data, &req); err != nil {
		c.sendError("invalid_data", "Invalid sync request")
		return
	}

	if req.Limit <= 0 || req.Limit > 100 {
		req.Limit = 100
	}

	messages, err := c.hub.db.GetMessagesSince(c.userID, req.SinceSeqID, req.Limit)
	if err != nil {
		log.Printf("Error getting messages: %v", err)
		c.sendError("db_error", "Failed to get messages")
		return
	}

	if len(messages) > 0 {
		lastSeq := messages[len(messages)-1].SeqID
		c.hub.db.UpdateSyncCursor(c.deviceID, lastSeq)
	}

	c.sendJSON(map[string]interface{}{
		"type": "sync_response",
		"data": map[string]interface{}{
			"messages":    messages,
			"has_more":    len(messages) == req.Limit,
			"server_time": time.Now().Unix(),
		},
	})
}

func (c *Client) handleAck(data json.RawMessage) {
	var req struct {
		SeqID int64 `json:"seq_id"`
	}

	if err := json.Unmarshal(data, &req); err != nil {
		return
	}

	message, err := c.hub.db.GetMessageBySeqID(req.SeqID)
	if err != nil || message == nil {
		return
	}

	c.hub.db.SetMessageStatus(message.ID, c.deviceID, "delivered")

	ackData, _ := json.Marshal(map[string]interface{}{
		"type": "status_update",
		"data": map[string]interface{}{
			"seq_id":    req.SeqID,
			"device_id": c.deviceID,
			"status":    "delivered",
		},
	})
	c.hub.SendToUser(message.SenderID, ackData, "")
}

func (c *Client) handleRead(data json.RawMessage) {
	var req struct {
		SeqID int64 `json:"seq_id"`
	}

	if err := json.Unmarshal(data, &req); err != nil {
		return
	}

	message, err := c.hub.db.GetMessageBySeqID(req.SeqID)
	if err != nil || message == nil {
		return
	}

	c.hub.db.SetMessageStatus(message.ID, c.deviceID, "read")

	readData, _ := json.Marshal(map[string]interface{}{
		"type": "status_update",
		"data": map[string]interface{}{
			"seq_id":    req.SeqID,
			"device_id": c.deviceID,
			"status":    "read",
		},
	})
	c.hub.SendToUser(message.SenderID, readData, "")
}

func (c *Client) handleDelete(data json.RawMessage) {
	var req struct {
		SeqID int64 `json:"seq_id"`
	}

	if err := json.Unmarshal(data, &req); err != nil {
		return
	}

	message, err := c.hub.db.GetMessageBySeqID(req.SeqID)
	if err != nil || message == nil {
		return
	}

	if message.SenderID != c.userID {
		c.sendError("forbidden", "Cannot delete other's messages")
		return
	}

	if err := c.hub.db.DeleteMessage(req.SeqID); err != nil {
		c.sendError("db_error", "Failed to delete message")
		return
	}

	deleteData, _ := json.Marshal(map[string]interface{}{
		"type": "message_deleted",
		"data": map[string]interface{}{
			"seq_id": req.SeqID,
		},
	})

	c.hub.SendToUser(c.userID, deleteData, "")
	c.hub.SendToUser(message.RecipientID, deleteData, "")

	log.Printf("🗑️ Message deleted: seq=%d by=%d", req.SeqID, c.userID)
}

func (c *Client) handlePing() {
	c.sendJSON(map[string]interface{}{
		"type": "pong",
		"data": map[string]interface{}{
			"server_time": time.Now().Unix(),
		},
	})
}

func (c *Client) sendJSON(v interface{}) {
	data, err := json.Marshal(v)
	if err != nil {
		return
	}

	select {
	case c.send <- data:
	default:
	}
}

func (c *Client) sendError(code, message string) {
	c.sendJSON(map[string]interface{}{
		"type": "error",
		"data": map[string]interface{}{
			"code":    code,
			"message": message,
		},
	})
}
```

## КОД: internal/db/db.go

```go
package db

import (
	"database/sql"
	"fmt"
	"os"
	"path/filepath"
	"time"

	_ "modernc.org/sqlite"
	"homelan-server/internal/models"
)

type Database struct {
	conn *sql.DB
}

func New(dbPath string) (*Database, error) {
	dir := filepath.Dir(dbPath)
	if err := os.MkdirAll(dir, 0700); err != nil {
		return nil, fmt.Errorf("failed to create db directory: %w", err)
	}

	conn, err := sql.Open("sqlite", dbPath)
	if err != nil {
		return nil, fmt.Errorf("failed to open database: %w", err)
	}

	if err := conn.Ping(); err != nil {
		return nil, fmt.Errorf("failed to ping database: %w", err)
	}

	_, err = conn.Exec("PRAGMA journal_mode=WAL; PRAGMA busy_timeout=5000;")
	if err != nil {
		return nil, fmt.Errorf("failed to set pragmas: %w", err)
	}

	db := &Database{conn: conn}

	if err := db.initSchema(); err != nil {
		return nil, fmt.Errorf("failed to init schema: %w", err)
	}

	return db, nil
}

func (db *Database) Close() error {
	return db.conn.Close()
}

func (db *Database) initSchema() error {
	schema := `
	CREATE TABLE IF NOT EXISTS users (
		id INTEGER PRIMARY KEY AUTOINCREMENT,
		username TEXT UNIQUE NOT NULL,
		password_hash TEXT NOT NULL,
		public_key BLOB,
		created_at INTEGER NOT NULL,
		last_seen INTEGER NOT NULL,
		is_admin INTEGER DEFAULT 0
	);

	CREATE TABLE IF NOT EXISTS devices (
		id INTEGER PRIMARY KEY AUTOINCREMENT,
		user_id INTEGER NOT NULL,
		device_id TEXT UNIQUE NOT NULL,
		device_name TEXT NOT NULL,
		device_type TEXT NOT NULL,
		public_key BLOB,
		created_at INTEGER NOT NULL,
		last_seen INTEGER NOT NULL,
		is_revoked INTEGER DEFAULT 0,
		FOREIGN KEY (user_id) REFERENCES users(id)
	);

	CREATE TABLE IF NOT EXISTS messages (
		id INTEGER PRIMARY KEY AUTOINCREMENT,
		seq_id INTEGER UNIQUE NOT NULL,
		sender_id INTEGER NOT NULL,
		recipient_id INTEGER NOT NULL,
		encrypted_content BLOB NOT NULL,
		content_type TEXT NOT NULL,
		file_id TEXT,
		created_at INTEGER NOT NULL,
		expires_at INTEGER,
		is_deleted INTEGER DEFAULT 0,
		deleted_at INTEGER,
		FOREIGN KEY (sender_id) REFERENCES users(id),
		FOREIGN KEY (recipient_id) REFERENCES users(id)
	);

	CREATE TABLE IF NOT EXISTS message_status (
		id INTEGER PRIMARY KEY AUTOINCREMENT,
		message_id INTEGER NOT NULL,
		device_id TEXT NOT NULL,
		status TEXT NOT NULL,
		updated_at INTEGER NOT NULL,
		FOREIGN KEY (message_id) REFERENCES messages(id),
		UNIQUE(message_id, device_id)
	);

	CREATE TABLE IF NOT EXISTS files (
		id TEXT PRIMARY KEY,
		uploader_id INTEGER NOT NULL,
		file_name_encrypted BLOB NOT NULL,
		file_size INTEGER NOT NULL,
		mime_type TEXT,
		storage_path TEXT NOT NULL,
		created_at INTEGER NOT NULL,
		expires_at INTEGER,
		is_deleted INTEGER DEFAULT 0,
		FOREIGN KEY (uploader_id) REFERENCES users(id)
	);

	CREATE TABLE IF NOT EXISTS sync_cursors (
		device_id TEXT PRIMARY KEY,
		last_seq_id INTEGER NOT NULL DEFAULT 0,
		updated_at INTEGER NOT NULL
	);

	CREATE INDEX IF NOT EXISTS idx_messages_seq ON messages(seq_id);
	CREATE INDEX IF NOT EXISTS idx_messages_sender ON messages(sender_id);
	CREATE INDEX IF NOT EXISTS idx_messages_recipient ON messages(recipient_id);
	CREATE INDEX IF NOT EXISTS idx_devices_user ON devices(user_id);
	`

	_, err := db.conn.Exec(schema)
	return err
}
```

```go
func (s *Server) handleWebSocket(w http.ResponseWriter, r *http.Request) {
	userID := r.URL.Query().Get("user_id")
	deviceID := r.URL.Query().Get("device_id")
	token := r.URL.Query().Get("token")

	if userID == "" || deviceID == "" || token == "" {
		http.Error(w, "Missing parameters", http.StatusBadRequest)
		return
	}

	device, err := s.db.GetDeviceByID(deviceID)
	if err != nil || device == nil {
		http.Error(w, "Device not found", http.StatusUnauthorized)
		return
	}
	if device.IsRevoked {
		http.Error(w, "Device revoked", http.StatusForbidden)
		return
	}

	conn, err := s.upgrader.Upgrade(w, r, nil)
	if err != nil {
		log.Printf("WebSocket upgrade error: %v", err)
		return
	}

	client := s.hub.RegisterClient(conn, device.UserID, deviceID)
	s.db.UpdateDeviceLastSeen(deviceID)
	s.db.UpdateUserLastSeen(device.UserID)

	go client.WritePump()
	go client.ReadPump()
}

func (s *Server) handleRegister(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	var req struct {
		Username   string `json:"username"`
		Password   string `json:"password"`
		DeviceID   string `json:"device_id"`
		DeviceName string `json:"device_name"`
		DeviceType string `json:"device_type"`
		PublicKey  []byte `json:"public_key"`
	}

	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "Invalid JSON", http.StatusBadRequest)
		return
	}

	if req.Username == "" || req.Password == "" || req.DeviceID == "" {
		http.Error(w, "Missing required fields", http.StatusBadRequest)
		return
	}

	existing, err := s.db.GetUserByUsername(req.Username)
	if err != nil {
		http.Error(w, "Internal error", http.StatusInternalServerError)
		return
	}
	if existing != nil {
		http.Error(w, "Username already exists", http.StatusConflict)
		return
	}

	admin, _ := s.db.GetAdmin()
	isAdmin := admin == nil

	user, err := s.db.CreateUser(req.Username, req.Password, isAdmin)
	if err != nil {
		http.Error(w, "Failed to create user", http.StatusInternalServerError)
		return
	}

	if len(req.PublicKey) > 0 {
		s.db.UpdateUserPublicKey(user.ID, req.PublicKey)
	}

	device, err := s.db.CreateDevice(user.ID, req.DeviceID, req.DeviceName, req.DeviceType, req.PublicKey)
	if err != nil {
		http.Error(w, "Failed to create device", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]interface{}{
		"success":     true,
		"user_id":     user.ID,
		"device_id":   device.DeviceID,
		"is_admin":    isAdmin,
		"server_time": time.Now().Unix(),
	})

	log.Printf("✅ User registered: %s (admin: %v)", req.Username, isAdmin)
}

func (s *Server) handleLogin(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	var req struct {
		Username string `json:"username"`
		Password string `json:"password"`
		DeviceID string `json:"device_id"`
	}

	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "Invalid JSON", http.StatusBadRequest)
		return
	}

	user, err := s.db.GetUserByUsername(req.Username)
	if err != nil {
		http.Error(w, "Internal error", http.StatusInternalServerError)
		return
	}
	if user == nil {
		http.Error(w, "Invalid credentials", http.StatusUnauthorized)
		return
	}

	if user.PasswordHash != req.Password {
		http.Error(w, "Invalid credentials", http.StatusUnauthorized)
		return
	}

	device, err := s.db.GetDeviceByID(req.DeviceID)
	if err != nil {
		http.Error(w, "Internal error", http.StatusInternalServerError)
		return
	}
	if device == nil {
		http.Error(w, "Device not registered", http.StatusUnauthorized)
		return
	}
	if device.IsRevoked {
		http.Error(w, "Device revoked", http.StatusForbidden)
		return
	}

	s.db.UpdateUserLastSeen(user.ID)
	s.db.UpdateDeviceLastSeen(req.DeviceID)

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]interface{}{
		"success":     true,
		"user_id":     user.ID,
		"is_admin":    user.IsAdmin,
		"server_time": time.Now().Unix(),
	})

	log.Printf("✅ User logged in: %s", req.Username)
}

func generateFileID() string {
	bytes := make([]byte, 16)
	rand.Read(bytes)
	return hex.EncodeToString(bytes)
}

func (s *Server) handleFileUpload(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodPost {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	userID := r.Header.Get("X-User-ID")
	if userID == "" {
		http.Error(w, "Unauthorized", http.StatusUnauthorized)
		return
	}

	r.Body = http.MaxBytesReader(w, r.Body, maxFileSize)

	if !utils.HasEnoughSpace(uploadPath, r.ContentLength) {
		http.Error(w, "Insufficient disk space", http.StatusInsufficientStorage)
		return
	}

	if err := os.MkdirAll(uploadPath, 0700); err != nil {
		http.Error(w, "Storage error", http.StatusInternalServerError)
		return
	}

	fileID := generateFileID()
	filePath := filepath.Join(uploadPath, fileID)

	dst, err := os.Create(filePath)
	if err != nil {
		http.Error(w, "Failed to create file", http.StatusInternalServerError)
		return
	}
	defer dst.Close()

	written, err := io.Copy(dst, io.LimitReader(r.Body, maxFileSize))
	if err != nil {
		os.Remove(filePath)
		http.Error(w, "Upload failed", http.StatusInternalServerError)
		return
	}

	userIDInt, _ := strconv.ParseInt(userID, 10, 64)
	fileName := r.Header.Get("X-File-Name")
	mimeType := r.Header.Get("Content-Type")
	if mimeType == "" {
		mimeType = "application/octet-stream"
	}

	err = s.db.CreateFile(fileID, userIDInt, []byte(fileName), written, mimeType, filePath)
	if err != nil {
		os.Remove(filePath)
		http.Error(w, "Database error", http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]interface{}{
		"success": true,
		"file_id": fileID,
		"size":    written,
	})

	log.Printf("📁 File uploaded: %s (%d bytes) by user %s", fileID, written, userID)
}

func (s *Server) handleFileDownload(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}

	fileID := filepath.Base(r.URL.Path)
	if fileID == "" || fileID == "files" {
		http.Error(w, "File ID required", http.StatusBadRequest)
		return
	}

	file, err := s.db.GetFileByID(fileID)
	if err != nil || file == nil {
		http.Error(w, "File not found", http.StatusNotFound)
		return
	}

	if file.IsDeleted {
		http.Error(w, "File deleted", http.StatusGone)
		return
	}

	f, err := os.Open(file.StoragePath)
	if err != nil {
		http.Error(w, "File not found", http.StatusNotFound)
		return
	}
	defer f.Close()

	stat, err := f.Stat()
	if err != nil {
		http.Error(w, "File error", http.StatusInternalServerError)
		return
	}

	fileSize := stat.Size()

	w.Header().Set("Content-Type", file.MimeType)
	w.Header().Set("Accept-Ranges", "bytes")

	rangeHeader := r.Header.Get("Range")
	if rangeHeader != "" {
		var start, end int64
		n, _ := fmt.Sscanf(rangeHeader, "bytes=%d-%d", &start, &end)
		if n == 1 {
			end = fileSize - 1
		}

		if start >= fileSize || end >= fileSize || start > end {
			http.Error(w, "Invalid range", http.StatusRequestedRangeNotSatisfiable)
			return
		}

		f.Seek(start, 0)
		length := end - start + 1

		w.Header().Set("Content-Range", fmt.Sprintf("bytes %d-%d/%d", start, end, fileSize))
		w.Header().Set("Content-Length", strconv.FormatInt(length, 10))
		w.WriteHeader(http.StatusPartialContent)

		io.CopyN(w, f, length)
	} else {
		w.Header().Set("Content-Length", strconv.FormatInt(fileSize, 10))
		io.Copy(w, f)
	}

	log.Printf("📥 File downloaded: %s", fileID)
}

func (s *Server) cleanupLoop() {
	ticker := time.NewTicker(1 * time.Hour)
	for range ticker.C {
		deleted, err := s.db.DeleteExpiredMessages()
		if err != nil {
			log.Printf("Error cleaning expired messages: %v", err)
		} else if deleted > 0 {
			log.Printf("🧹 Cleaned %d expired messages", deleted)
		}

		paths, err := s.db.DeleteExpiredFiles()
		if err != nil {
			log.Printf("Error cleaning expired files: %v", err)
		} else {
			for _, path := range paths {
				if err := os.Remove(path); err != nil {
					log.Printf("Error removing file %s: %v", path, err)
				}
			}
			if len(paths) > 0 {
				log.Printf("🧹 Cleaned %d expired files", len(paths))
			}
		}
	}
}

func (s *Server) deadManSwitchLoop() {
	ticker := time.NewTicker(1 * time.Hour)
	for range ticker.C {
		if time.Now().Year() < 2024 {
			log.Printf("⚠️ Invalid system time, skipping dead man's switch check")
			continue
		}

		admin, err := s.db.GetAdmin()
		if err != nil || admin == nil {
			continue
		}

		daysSinceLastSeen := time.Since(time.Unix(admin.LastSeen, 0)).Hours() / 24
		if daysSinceLastSeen > 30 {
			log.Printf("💀 Dead Man's Switch triggered! Admin not seen for %.0f days", daysSinceLastSeen)
			s.triggerPanic()
		}
	}
}

func (s *Server) triggerPanic() {
	log.Printf("🚨 PANIC WIPE INITIATED")

	s.db.Close()

	dbPath := s.config.Database.Path
	os.Remove(dbPath + "-wal")
	os.Remove(dbPath + "-shm")

	f, err := os.OpenFile(dbPath, os.O_WRONLY, 0)
	if err == nil {
		garbage := make([]byte, 4096)
		rand.Read(garbage)
		f.Write(garbage)
		f.Sync()
		f.Close()
	}

	os.RemoveAll(uploadPath)

	log.Printf("🚨 PANIC WIPE COMPLETE")
	os.Exit(1)
}

func (s *Server) Close() error {
	return s.db.Close()
}

	}
}
```
