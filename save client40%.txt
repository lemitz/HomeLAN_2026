```dart
import 'dart:convert';
import 'dart:typed_data';

class Message {
  final int? id;
  final int seqId;
  final int senderId;
  final int recipientId;
  final Uint8List encryptedContent;
  final String contentType;
  final String? fileId;
  final int createdAt;
  final int? expiresAt;
  final bool isDeleted;
  final String status;

  Message({
    this.id,
    required this.seqId,
    required this.senderId,
    required this.recipientId,
    required this.encryptedContent,
    this.contentType = 'text',
    this.fileId,
    required this.createdAt,
    this.expiresAt,
    this.isDeleted = false,
    this.status = 'pending',
  });

  String get textContent {
    try {
      return utf8.decode(encryptedContent);
    } catch (e) {
      return base64Encode(encryptedContent);
    }
  }

  factory Message.fromJson(Map<String, dynamic> json) {
    Uint8List content;
    if (json['encrypted_content'] is String) {
      content = base64Decode(json['encrypted_content']);
    } else if (json['encrypted_content'] is List) {
      content = Uint8List.fromList(List<int>.from(json['encrypted_content']));
    } else {
      content = Uint8List(0);
    }

    return Message(
      id: json['id'],
      seqId: json['seq_id'] ?? 0,
      senderId: json['sender_id'] ?? 0,
      recipientId: json['recipient_id'] ?? 0,
      encryptedContent: content,
      contentType: json['content_type'] ?? 'text',
      fileId: json['file_id'],
      createdAt: json['created_at'] ?? 0,
      expiresAt: json['expires_at'],
      isDeleted: json['is_deleted'] ?? false,
      status: json['status'] ?? 'sent',
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'id': id,
      'seq_id': seqId,
      'sender_id': senderId,
      'recipient_id': recipientId,
      'encrypted_content': base64Encode(encryptedContent),
      'content_type': contentType,
      'file_id': fileId,
      'created_at': createdAt,
      'expires_at': expiresAt,
      'is_deleted': isDeleted,
      'status': status,
    };
  }

  Message copyWith({
    int? id,
    int? seqId,
    String? status,
    bool? isDeleted,
  }) {
    return Message(
      id: id ?? this.id,
      seqId: seqId ?? this.seqId,
      senderId: senderId,
      recipientId: recipientId,
      encryptedContent: encryptedContent,
      contentType: contentType,
      fileId: fileId,
      createdAt: createdAt,
      expiresAt: expiresAt,
      isDeleted: isDeleted ?? this.isDeleted,
      status: status ?? this.status,
    );
  }
}
```

```dart
class ApiService {

    required String password,
    required String deviceId,
    required String deviceName,

        'username': username,
        'password': password,
        'device_id': deviceId,
        'device_name': deviceName,
        'device_type': deviceType,
      }),
    );

    if (response.statusCode == 200) {
      return jsonDecode(response.body);
    } else {
      throw ApiException(response.statusCode, response.body);
    }
  }

  Future<Map<String, dynamic>> login({
    required String username,
    required String password,
    required String deviceId,
  }) async {
    final response = await http.post(
      Uri.parse('$baseUrl/api/login'),
      headers: {'Content-Type': 'application/json'},
      body: jsonEncode({
        'username': username,
        'password': password,
        'device_id': deviceId,
      }),
    );

    if (response.statusCode == 200) {
      return jsonDecode(response.body);
    } else {
      throw ApiException(response.statusCode, response.body);
    }
  }

  Future<Map<String, dynamic>> getHealth() async {
    final response = await http.get(Uri.parse('$baseUrl/health'));
    if (response.statusCode == 200) {
      return jsonDecode(response.body);
    } else {
      throw ApiException(response.statusCode, response.body);
    }
  }

  Future<int> getServerTime() async {
    final response = await http.get(Uri.parse('$baseUrl/api/time'));
    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      return data['server_time'] ?? 0;
    }
    return 0;
  }
}

}
```

```dart
import 'dart:async';
import 'dart:convert';
import 'package:web_socket_channel/web_socket_channel.dart';

class WebSocketService {
  final String baseUrl;
  WebSocketChannel? _channel;
  
  final _messageController = StreamController<Map<String, dynamic>>.broadcast();
  Stream<Map<String, dynamic>> get messages => _messageController.stream;
  
  bool _isConnected = false;
  bool get isConnected => _isConnected;
  
  Timer? _pingTimer;
  Timer? _reconnectTimer;
  
  int? _userId;
  String? _deviceId;
  String? _token;

  WebSocketService({required this.baseUrl});

  Future<void> connect({
    required int userId,
    required String deviceId,
    required String token,
  }) async {
    _userId = userId;
    _deviceId = deviceId;
    _token = token;
    
    await _doConnect();
  }

  Future<void> _doConnect() async {
    if (_userId == null || _deviceId == null) return;
    
    try {
      final wsUrl = baseUrl.replaceFirst('http', 'ws');
      final uri = Uri.parse('$wsUrl/ws?user_id=$_userId&device_id=$_deviceId&token=$_token');
      
      _channel = WebSocketChannel.connect(uri);
      
      _channel!.stream.listen(
        (data) {
          _isConnected = true;
          final message = jsonDecode(data as String);
          _messageController.add(message);
        },
        onError: (error) {
          print('WebSocket error: $error');
          _isConnected = false;
          _scheduleReconnect();
        },
        onDone: () {
          print('WebSocket closed');
          _isConnected = false;
          _scheduleReconnect();
        },
      );
      
      _isConnected = true;
      _startPingTimer();
      
      print('WebSocket connected');
    } catch (e) {
      print('WebSocket connection error: $e');
      _isConnected = false;
      _scheduleReconnect();
    }
  }

  void _startPingTimer() {
    _pingTimer?.cancel();
    _pingTimer = Timer.periodic(const Duration(seconds: 25), (_) {
      if (_isConnected) {
        send({'type': 'ping', 'data': {}});
      }
    });
  }

  void _scheduleReconnect() {
    _reconnectTimer?.cancel();
    _reconnectTimer = Timer(const Duration(seconds: 5), () {
      if (!_isConnected && _userId != null) {
        print('Reconnecting...');
        _doConnect();
      }
    });
  }

  void send(Map<String, dynamic> message) {
    if (_channel != null && _isConnected) {
      _channel!.sink.add(jsonEncode(message));
    }
  }

  void sendMessage({
    required int recipientId,
    required String content,
    String contentType = 'text',
    String? fileId,
  }) {
    send({
      'type': 'send_message',
      'data': {
        'recipient_id': recipientId,
        'encrypted_content': base64Encode(utf8.encode(content)),
        'content_type': contentType,
        if (fileId != null) 'file_id': fileId,
      },
    });
  }

  void sync({int sinceSeqId = 0, int limit = 100}) {
    send({
      'type': 'sync',
      'data': {
        'since_seq_id': sinceSeqId,
        'limit': limit,
      },
    });
  }

  void ack(int seqId) {
    send({
      'type': 'ack',
      'data': {'seq_id': seqId},
    });
  }

  void markRead(int seqId) {
    send({
      'type': 'read',
      'data': {'seq_id': seqId},
    });
  }

  void deleteMessage(int seqId) {
    send({
      'type': 'delete',
      'data': {'seq_id': seqId},
    });
  }

  void disconnect() {
    _pingTimer?.cancel();
    _reconnectTimer?.cancel();
    _channel?.sink.close();
    _isConnected = false;
  }

  void dispose() {
    disconnect();
    _messageController.close();
  }
}
```

```dart
import 'dart:io';
import 'package:path/path.dart';
import 'package:path_provider/path_provider.dart';
import 'package:sqflite_common_ffi/sqflite_ffi.dart';
import '../models/models.dart';

class DatabaseService {
  static Database? _database;

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDB();
    return _database!;
  }

  Future<Database> _initDB() async {
    if (Platform.isWindows || Platform.isLinux) {
      sqfliteFfiInit();
      databaseFactory = databaseFactoryFfi;
    }

    final directory = await getApplicationDocumentsDirectory();
    final path = join(directory.path, 'homelan_client.db');

    return await openDatabase(
      path,
      version: 1,
      onCreate: _onCreate,
    );
  }

  Future<void> _onCreate(Database db, int version) async {
    await db.execute('''
      CREATE TABLE messages (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        seq_id INTEGER UNIQUE,
        sender_id INTEGER NOT NULL,
        recipient_id INTEGER NOT NULL,
        encrypted_content BLOB NOT NULL,
        content_type TEXT NOT NULL,
        file_id TEXT,
        created_at INTEGER NOT NULL,
        expires_at INTEGER,
        is_deleted INTEGER DEFAULT 0,
        status TEXT DEFAULT 'sent'
      )
    ''');

    await db.execute('''
      CREATE TABLE users (
        id INTEGER PRIMARY KEY,
        username TEXT NOT NULL,
        is_admin INTEGER DEFAULT 0,
        created_at INTEGER,
        last_seen INTEGER
      )
    ''');

    await db.execute('''
      CREATE TABLE settings (
        key TEXT PRIMARY KEY,
        value TEXT
      )
    ''');

    await db.execute('CREATE INDEX idx_messages_seq ON messages(seq_id)');
    await db.execute('CREATE INDEX idx_messages_sender ON messages(sender_id)');
    await db.execute('CREATE INDEX idx_messages_recipient ON messages(recipient_id)');
  }

  Future<int> insertMessage(Message message) async {
    final db = await database;
    return await db.insert(
      'messages',
      {
        'seq_id': message.seqId,
        'sender_id': message.senderId,
        'recipient_id': message.recipientId,
        'encrypted_content': message.encryptedContent,
        'content_type': message.contentType,
        'file_id': message.fileId,
        'created_at': message.createdAt,
        'expires_at': message.expiresAt,
        'is_deleted': message.isDeleted ? 1 : 0,
        'status': message.status,
      },
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }

  Future<List<Message>> getMessages(int oderId, int myId, {int limit = 50}) async {
    final db = await database;
    final List<Map<String, dynamic>> maps = await db.query(
      'messages',
      where: '((sender_id = ? AND recipient_id = ?) OR (sender_id = ? AND recipient_id = ?)) AND is_deleted = 0',
      whereArgs: [myId, oderId, oderId, myId],
      orderBy: 'created_at DESC',
      limit: limit,
    );

    return List.generate(maps.length, (i) {
      return Message(
        id: maps[i]['id'],
        seqId: maps[i]['seq_id'] ?? 0,
        senderId: maps[i]['sender_id'],
        recipientId: maps[i]['recipient_id'],
        encryptedContent: maps[i]['encrypted_content'],
        contentType: maps[i]['content_type'] ?? 'text',
        fileId: maps[i]['file_id'],
        createdAt: maps[i]['created_at'],
        expiresAt: maps[i]['expires_at'],
        isDeleted: maps[i]['is_deleted'] == 1,
        status: maps[i]['status'] ?? 'sent',
      );
    });
  }

  Future<int> getLastSeqId() async {
    final db = await database;
    final result = await db.rawQuery('SELECT MAX(seq_id) as max_seq FROM messages');
    return result.first['max_seq'] as int? ?? 0;
  }

  Future<void> updateMessageStatus(int seqId, String status) async {
    final db = await database;
    await db.update(
      'messages',
      {'status': status},
      where: 'seq_id = ?',
      whereArgs: [seqId],
    );
  }

  Future<void> deleteMessage(int seqId) async {
    final db = await database;
    await db.update(
      'messages',
      {'is_deleted': 1},
      where: 'seq_id = ?',
      whereArgs: [seqId],
    );
  }

  Future<void> insertUser(User user) async {
    final db = await database;
    await db.insert(
      'users',
      user.toJson(),
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }

  Future<User?> getUser(int id) async {
    final db = await database;
    final maps = await db.query('users', where: 'id = ?', whereArgs: [id]);
    if (maps.isEmpty) return null;
    return User.fromJson(maps.first);
  }

  Future<List<User>> getAllUsers() async {
    final db = await database;
    final maps = await db.query('users');
    return maps.map((m) => User.fromJson(m)).toList();
  }

  Future<void> setSetting(String key, String value) async {
    final db = await database;
    await db.insert(

  }

  Future<void> clearAll() async {
    final db = await database;
    await db.delete('messages');
    await db.delete('users');
    await db.delete('settings');
  }
}
```

```dart
class AuthProvider with ChangeNotifier {
  final ApiService _api;
  final WebSocketService _ws;

  int? _userId;
  String? _username;
  String? _deviceId;
  bool _isAdmin = false;
  bool _isLoading = false;

  bool get isAdmin => _isAdmin;

  bool get isLoggedIn => _userId != null;

  AuthProvider({
    required ApiService api,
    required WebSocketService ws,
    required DatabaseService db,
  })  : _api = api,
        _ws = ws,
        _db = db;

  Future<void> init() async {
    final prefs = await SharedPreferences.getInstance();
    _userId = prefs.getInt('user_id');
    _username = prefs.getString('username');
    _deviceId = prefs.getString('device_id');
    _isAdmin = prefs.getBool('is_admin') ?? false;

    if (_deviceId == null) {
      _deviceId = const Uuid().v4();
      await prefs.setString('device_id', _deviceId!);
    }

    if (_userId != null && _deviceId != null) {
      await _connectWebSocket();
    }

    notifyListeners();
  }

  Future<bool> register(String username, String password) async {
    _isLoading = true;
    _error = null;
    notifyListeners();

    try {
      final deviceName = Platform.localHostname;
      final deviceType = Platform.operatingSystem;

      final result = await _api.register(
        username: username,
        password: password,
        deviceId: _deviceId!,
        deviceName: deviceName,
        deviceType: deviceType,
      );

      _userId = result['user_id'];
      _username = username;
      _isAdmin = result['is_admin'] ?? false;

      await _saveSession();
      await _connectWebSocket();

      _isLoading = false;
      notifyListeners();
      return true;
    } catch (e) {
      _error = e.toString();
      _isLoading = false;
      notifyListeners();
      return false;
    }
  }

  Future<bool> login(String username, String password) async {
    _isLoading = true;
    _error = null;
    notifyListeners();

    try {
      final result = await _api.login(
        username: username,
        password: password,
        deviceId: _deviceId!,
      );

      _userId = result['user_id'];
      _username = username;
      _isAdmin = result['is_admin'] ?? false;

      await _saveSession();
      await _connectWebSocket();

      _isLoading = false;
      notifyListeners();
      return true;
    } catch (e) {
      _error = e.toString();
      _isLoading = false;
      notifyListeners();
      return false;
    }
  }

  Future<void> logout() async {
    _ws.disconnect();

    final prefs = await SharedPreferences.getInstance();
    await prefs.remove('user_id');
    await prefs.remove('username');
    await prefs.remove('is_admin');

    _userId = null;
    _username = null;
    _isAdmin = false;

    notifyListeners();
  }

  Future<void> _saveSession() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setInt('user_id', _userId!);
    await prefs.setString('username', _username!);
    await prefs.setBool('is_admin', _isAdmin);
  }

  Future<void> _connectWebSocket() async {
    if (_userId != null && _deviceId != null) {
      await _ws.connect(
        userId: _userId!,
        deviceId: _deviceId!,
        token: 'temp_token',
      );
    }
  }

  void clearError() {
    _error = null;
    notifyListeners();
  }
}
```

```dart
import 'dart:async';
import 'dart:convert';
import 'package:flutter/foundation.dart';
import '../models/models.dart';
import '../services/services.dart';

class ChatProvider with ChangeNotifier {
  final WebSocketService _ws;
  final DatabaseService _db;
  
  int? _myUserId;
  final Map<int, List<Message>> _chats = {};
  final Map<int, String> _usernames = {};
  StreamSubscription? _wsSubscription;
  bool _isLoading = false;

  Map<int, List<Message>> get chats => _chats;
  Map<int, String> get usernames => _usernames;
  bool get isLoading => _isLoading;

  ChatProvider({
    required WebSocketService ws,
    required DatabaseService db,
  })  : _ws = ws,
        _db = db {
    _listenToWebSocket();
  }

  void setUserId(int userId) {
    _myUserId = userId;
    _syncMessages();
  }

  void _listenToWebSocket() {
    _wsSubscription = _ws.messages.listen((message) {
      final type = message['type'];
      final data = message['data'];

      switch (type) {
        case 'new_message':
          _handleNewMessage(data);
          break;
        case 'message_sent':
          _handleMessageSent(data);
          break;
        case 'sync_response':
          _handleSyncResponse(data);
          break;
        case 'status_update':
          _handleStatusUpdate(data);
          break;
        case 'message_deleted':
          _handleMessageDeleted(data);
          break;
      }
    });
  }

  void _handleNewMessage(Map<String, dynamic> data) {
    final message = Message.fromJson(data);
    
    final oderId = message.senderId == _myUserId 
        ? message.recipientId 
        : message.senderId;
    
    _chats[oderId] ??= [];
    
    final exists = _chats[oderId]!.any((m) => m.seqId == message.seqId);
    if (!exists) {
      _chats[oderId]!.insert(0, message);
      _db.insertMessage(message);
      
      _ws.ack(message.seqId);
    }
    
    notifyListeners();
  }

  void _handleMessageSent(Map<String, dynamic> data) {
    final seqId = data['seq_id'] as int;
    final createdAt = data['created_at'] as int;
    
    for (var messages in _chats.values) {
      for (var i = 0; i < messages.length; i++) {
        if (messages[i].status == 'pending' && messages[i].seqId == 0) {
          messages[i] = messages[i].copyWith(
            seqId: seqId,
            status: 'sent',
          );
          _db.insertMessage(messages[i]);
          break;
        }
      }
    }
    
    notifyListeners();
  }

  void _handleSyncResponse(Map<String, dynamic> data) {
    final messages = (data['messages'] as List?)
        ?.map((m) => Message.fromJson(m))
        .toList() ?? [];
    
    for (var message in messages) {
      final oderId = message.senderId == _myUserId 
          ? message.recipientId 
          : message.senderId;
      
      _chats[oderId] ??= [];
      
      final exists = _chats[oderId]!.any((m) => m.seqId == message.seqId);
      if (!exists) {
        _chats[oderId]!.add(message);
        _db.insertMessage(message);
      }
    }
    
    for (var oderId in _chats.keys) {
      _chats[oderId]!.sort((a, b) => b.createdAt.compareTo(a.createdAt));
    }
    
    _isLoading = false;
    notifyListeners();
  }

  void _handleStatusUpdate(Map<String, dynamic> data) {
    final seqId = data['seq_id'] as int;
    final status = data['status'] as String;
    
    for (var messages in _chats.values) {
      for (var i = 0; i < messages.length; i++) {
        if (messages[i].seqId == seqId) {
          messages[i] = messages[i].copyWith(status: status);
          _db.updateMessageStatus(seqId, status);
          break;
        }
      }
    }
    
    notifyListeners();
  }

  void _handleMessageDeleted(Map<String, dynamic> data) {
    final seqId = data['seq_id'] as int;
    
    for (var messages in _chats.values) {
      messages.removeWhere((m) => m.seqId == seqId);
    }
    
    _db.deleteMessage(seqId);
    notifyListeners();
  }

  Future<void> _syncMessages() async {
    if (_myUserId == null) return;
    
    _isLoading = true;
    notifyListeners();
    
    final lastSeqId = await _db.getLastSeqId();
    _ws.sync(sinceSeqId: lastSeqId);
  }

  Future<void> loadLocalMessages(int oderId) async {
    if (_myUserId == null) return;
    
    final messages = await _db.getMessages(oderId, _myUserId!);
    _chats[oderId] = messages;
    notifyListeners();
  }

  void sendMessage(int recipientId, String content) {
    if (_myUserId == null) return;
    
    final tempMessage = Message(
      seqId: 0,
      senderId: _myUserId!,
      recipientId: recipientId,
      encryptedContent: Uint8List.fromList(utf8.encode(content)),
      contentType: 'text',
      createdAt: DateTime.now().millisecondsSinceEpoch ~/ 1000,
      status: 'pending',
    );
    
    _chats[recipientId] ??= [];
    _chats[recipientId]!.insert(0, tempMessage);
    notifyListeners();
    
    _ws.sendMessage(recipientId: recipientId, content: content);
  }

  void deleteMessage(int seqId) {
    _ws.deleteMessage(seqId);
  }

  void markAsRead(int seqId) {
    _ws.markRead(seqId);
  }

  void addUser(int userId, String username) {
    _usernames[userId] = username;
    notifyListeners();
  }

  List<int> getChatUserIds() {
    return _chats.keys.toList();
  }

  @override
  void dispose() {
    _wsSubscription?.cancel();
    super.dispose();
  }
}
```
